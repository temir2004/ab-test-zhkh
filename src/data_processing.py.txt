import pandas as pd
import os

def load_and_clean_data(raw_data_path: str) -> pd.DataFrame:
    """Загружает данные, очищает от выбросов и объединяет таблицы."""
    print("Загрузка данных...")
    users = pd.read_excel(raw_data_path, sheet_name='Пользователи')
    payments = pd.read_excel(raw_data_path, sheet_name='Платежи')

    print("Очистка аномалий (EDA)...")
    # Удаляем пользователей с некорректными данными (возраст, пустой город, без группы)
    bad_user_ids = [9001, 9002, 9003, 9004]
    users_clean = users[~users['user_id'].isin(bad_user_ids)]

    # Удаляем платежи без юзера и с несовпадением групп
    bad_payment_ids = [99003, 99002, 7658]
    payments_clean = payments[~payments['payment_id'].isin(bad_payment_ids)]

    # Мерджим таблицы
    df = payments_clean.merge(users_clean[['user_id', 'group']], on='user_id', how='inner')
    print(f"Данные готовы. Уникальных попыток оплаты: {len(df)}")
    
    return df

def save_processed_data(df: pd.DataFrame, output_path: str):
    """Сохраняет очищенные данные в CSV."""
    os.makedirs(os.path.dirname(output_path), exist_ok=True)
    df.to_csv(output_path, index=False)
    print(f"Очищенные данные сохранены в {output_path}")
